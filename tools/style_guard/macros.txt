# Pezzi di espressione riutilizzabili da tutte le liste.
# Una riga "@NOME = ..." definisce; nelle liste si richiama scrivendo @NOME.
# Questo file viene caricato prima di ogni lista, cosi' la stessa macro non va
# ripetuta in tre file diversi (e non puo' divergere fra l'uno e l'altro).
# Una macro puo' usarne un'altra: l'espansione continua finche' non restano @.

# Articolo o dimostrativo.
@DET = (un|uno|una|un'|il|lo|la|i|gli|le|dei|degli|delle|questo|questa|questi|queste|quel|quella|quello)

# Gruppo nominale minimo: articolo piu' un sostantivo.
# Serve per "e' un'idea scarsa", dove l'aggettivo non tocca il verbo.
@NOUN = @DET\s*[a-zàèéìòùA-ZÀÈÉÌÒÙ]+

# Forma predicativa: il verbo copulativo davanti all'aggettivo, con in mezzo un
# gruppo nominale o un avverbio di grado, entrambi facoltativi.
# E' cio' che separa un giudizio da una descrizione: "e' solido" giudica,
# "piu' solido di un'opinione" no.
@PRED = (è|e'|sono|era|erano|sembra|sembrano|resta|restano|risulta|risultano)\s+(@NOUN\s+)?(molto\s+|davvero\s+|piuttosto\s+|abbastanza\s+)?

# Forma predicativa inglese: stesso lavoro di @PRED sulla meta' del progetto che
# resta in inglese, cioe' TDD.md e i messaggi di commit.
@DETEN = (a|an|the|this|that|these|those)
@NOUNEN = @DETEN\s+[a-zA-Z]+
@PREDEN = (is|are|was|were|seems|seem|looks|look|feels|remains)\s+(@NOUNEN\s+)?(very\s+|really\s+|quite\s+|pretty\s+)?
